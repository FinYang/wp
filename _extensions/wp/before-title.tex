%% CAPTIONS
\usepackage{caption}
% Make the label (e.g. Figure 1) bold, the text italic, and a colon to separate them
\DeclareCaptionStyle{italic}[justification=centering]
{labelfont={bf},textfont={it},labelsep=colon}
% Apply this to figures and tables.
% Multiline caption hang under the caption text
% rather than putting subsequent lines underneath the label (more indent)
%
\captionsetup[figure]{style=italic,format=hang,singlelinecheck=true}
\captionsetup[table]{style=italic,format=hang,singlelinecheck=true}

%% FONT
\usepackage{bera}
\usepackage[charter]{mathdesign}
%\usepackage[scale=0.9]{sourcecodepro}
\usepackage[lf,t]{FiraSans}

%% HEADERS AND FOOTERS
\usepackage{fancyhdr}
\pagestyle{fancy}
% Page number in the right footer
% larger, bold
\rfoot{\Large\sffamily\raisebox{-0.1cm}{\textbf{\thepage}}}
% Title in the left header
\makeatletter
\lhead{\textsf{\expandafter{\@title}}}
\makeatother
% Empty right header and centre footer
\rhead{}
\cfoot{}
\setlength{\headheight}{15pt}
% Add horizontal lines below the header and above the footer
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0.4pt}
% Special treatment for the plain page style
% usually the first page of a chapter/document
\fancypagestyle{plain}{%
  \fancyhf{} % clear all header and footer fields
  \fancyfoot[C]{\sffamily\thepage} % except the center
  % Remove lines around headers and footers
  \renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}}

%% MATHS
\usepackage{bm,amsmath}
% Allow equations to break across pages
\allowdisplaybreaks

%% GRAPHICS
% Default placement options for figures
\makeatletter
\def\fps@figure{htbp}
\makeatother
% Maximum 2 floats at the top of a page
\setcounter{topnumber}{2}
% Maximum 2 floats at the bottom
\setcounter{bottomnumber}{2}
% Maximum 4 floats in total on a text page
\setcounter{totalnumber}{4}
% Up to 85% of a page can be occupied by floats at the top
\renewcommand{\topfraction}{0.85}
% Up to 85% of a page can be occupied by floats at the bottom
\renewcommand{\bottomfraction}{0.85}
% At least 15% of the page should normally be text.
\renewcommand{\textfraction}{0.15}
% If page containing only floats, they should occupy at least 80% of the page.
\renewcommand{\floatpagefraction}{0.8}

%% SECTION TITLES
\usepackage[compact,sf,bf]{titlesec}
\titleformat{\section}[block]
{\fontsize{15}{17}\bfseries\sffamily}
{\thesection}
{0.4em}{} % puts 0.4 em of space between the number and title
\titleformat{\subsection}[block]
{\fontsize{12}{14}\bfseries\sffamily}
{\thesubsection}
{0.4em}{}
%   left indentation, spacing before, spacing after
\titlespacing{\section}{0pt}{*5}{*1}
\titlespacing{\subsection}{0pt}{*2}{*0.2}
\titlespacing{\subsubsection}{0pt}{*1}{*0.1}
